<script src="https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2"></script>



<script>


// =====================
// Supabase
// =====================


const SUPABASE_URL =
"https://zorgenrnjbwnzsnwalek.supabase.co";


const SUPABASE_KEY =
"ใส่ key เดียวกับ index.html";



const client =
supabase.createClient(
SUPABASE_URL,
SUPABASE_KEY
);




// ตรวจสอบสถานะ


async function checkStatus(){


let id =
document.getElementById("studentID")
.value
.trim();



if(id===""){

alert("กรุณากรอกรหัสนักศึกษา");

return;

}




const {data,error}=await client

.from("problems")

.select("*")

.eq(
"student_id",
id
)

.order(
"created_at",
{
ascending:false
}
);




if(error){

alert(error.message);

return;

}




let html="";



if(data.length===0){


html=`

<div class="card">

ไม่พบข้อมูลการแจ้งปัญหา

</div>

`;


}

else{


data.forEach(item=>{


html += `


<div class="card">


<p>
ประเภทปัญหา:
${item.category}
</p>


<p>
รายละเอียด:
${item.detail}
</p>


<p>
สถานะ:

<span class="status">

${item.status}

</span>

</p>


</div>


`;


});


}



document.getElementById("result")
.innerHTML=html;



}



</script>
